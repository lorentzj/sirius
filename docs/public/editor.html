fn dot{N}(a: f32[N], b: f32[N]) -> f32:
    let mut sum = 0.0
    for i from 0 to N:
        sum += a[i] * b[i]
    return sum

fn matmul{I, J, L}(a: f32[I, J], b: f32[J, L]) -> f32[I, L]:
    for i from 0 to I:
        for k from 0 to L:
            yield dot(a[i], b'[k])
