9/// Closed-form sum of a [Poly] over given bounds.
10///
11/// Given the code:
12/// ```text
13/// for i from 0 to N:
14///     for j from 0 to i^2:
15///         yield 1
16/// ```
17///
18/// The inner `for` body `yield`s $1$ time, outer `for` body `yield`s $\sum_{j=0}^{i^2-1} (1) = i^2$ times, and the full program `yield`s $\sum_{i=0}^{N-1} (i^2) = N(N-1)(2N - 1)/6$ times.
19///
20/// To count `yield`s in general, we need an algorithm to count lattice points over [`Poly`] bounds. Treating other [`Var`]s as constants, and summing each term independently, the problem reduces to
21/// $\sum_{x=0}^{P-1} x^{n}$ for variable $x$, constant $n$, and polynomial $P$.
22///
23/// We can convert to [falling factorials](https://en.wikipedia.org/wiki/Falling_and_rising_factorials) $x_{(k)} = (x)(x-1)(x-2)\dots(x-k+1)$ using [Stirling numbers of the second kind](https://en.wikipedia.org/wiki/Stirling_numbers_of_the_second_kind) and the identity
24///
25/// $$x^{n} = \sum_{k=0}^n {S(n, k)} x_{(k)}$$
26///
27/// Therefore, $$\sum_{x=0}^{P-1} x^n = \sum_{x=0}^{P-1} \left( \sum_{k=0}^n S(n, k) x_{(k)} \right)$$
28///
29/// After the inner sum is simplified, each term will be a falling factorial that can be evaluated with the discrete power rule
30///
31/// $$\sum_{x=0}^{K-1} x_{(k)} = \frac{K_{(k+1)}}{k+1}$$
32///
33/// For example, in the case above, $P=N$ and $n=2$, so
34/// $$\sum_{x=0}^{N-1} x^2 = \sum_{x=0}^{N-1} \left( \sum_{k=0}^2 \textcolor{blue}{S(2, k)} \textcolor{red}{x_{(k)}} \right)$$
35/// $$= \sum_{x=0}^{N-1} \textcolor{blue}{(0)}\textcolor{red}{x_{(0)}} + \sum_{x=0}^{N-1} \textcolor{blue}{(1)}\textcolor{red}{x_{(1)}} + \sum_{x=0}^{N-1} \textcolor{blue}{
36/// (1)}\textcolor{red}{x_{(2)}}$$
37/// $$= \frac{N\_{(2)}}{2} + \frac{N\_{(3)}}{3}$$
38/// $$= \frac{N(N-1)}{2} + \frac{N(N-1)(N-2)}{3}$$
39/// $$= \frac{3(N^2-N) + 2(N^3-3N^2+2N)}{6}$$
40/// $$= \frac{2N^3 - 3N^2 + N}{6}$$
41/// $$ = \frac{N(N−1)(2N−1)}{6}$$
42///
43/// This algorithm is implemented in [`Count::sum_below`].
44#[derive(Clone, PartialEq, Eq, Hash, Debug)]
45pub struct Count {
46    num: Poly,
47    den: Coef,
48}
49
50impl Count {
51    pub fn zero() -> Self {
52        Self {
53            num: Poly::zero(),
54            den: 1.into(),
55        }
56    }
57
58    pub fn constant<T: Into<Coef>>(c: T) -> Self {
59        Self {
60            num: Poly::constant(c),
61            den: 1.into(),
62        }
63    }
64
65    pub fn var(v: Var, p: Pow) -> Self {
66        Self {
67            num: Poly::var(v, p),
68            den: 1.into(),
69        }
70    }
71
72    pub fn new<T: Into<Coef>>(num: Poly, den: T) -> Self {
73        let den = den.into();
74        assert!(!den.is_zero());
75
76        let (num, den) = if !den.is_positive() {
77            (num.mul_scalar(-1), -den)
78        } else {
79            (num, den)
80        };
81        let g = num.coef_gcd().gcd(&den);
82        let num = num.divide_coefs(g).unwrap();
83
84        Self { num, den: den / g }.debug_checked()
85    }
86
87    pub fn num(&self) -> &Poly {
88        &self.num
89    }
90
91    pub fn den(&self) -> Coef {
92        self.den
93    }
94
95    pub fn is_zero(&self) -> bool {
96        self.num.is_zero()
97    }
98
99    /// Only possible if denominator is zero.
100    pub fn as_poly(&self) -> Option<&Poly> {
101        (self.den == 1.into()).then_some(&self.num)
102    }
103
104    pub fn as_constant(&self) -> Option<(Coef, Coef)> {
105        self.num.as_constant().map(|c| (c, self.den))
106    }
107
108    pub fn add(&self, rhs: &Self) -> Self {
109        let g = self.den.gcd(&rhs.den);
110        let lcm = self.den / g * rhs.den;
111        let num = self
112            .num
113            .mul_scalar(lcm / self.den)
114            .add(&rhs.num.mul_scalar(lcm / rhs.den));
115        Self::new(num, lcm).debug_checked()
116    }
117
118    pub fn neg(&self) -> Self {
119        Self {
120            num: self.num.neg(),
121            den: self.den,
122        }
123        .debug_checked()
124    }
125
126    pub fn sub(&self, rhs: &Self) -> Self {
127        self.add(&rhs.neg())
128    }
129
130    pub fn mul(&self, rhs: &Self) -> Self {
131        Self::new(self.num.mul(&rhs.num), self.den * rhs.den)
132    }
133
134    pub fn mul_scalar<T: Into<Coef>>(&self, c: T) -> Self {
135        Self::new(self.num.mul_scalar(c), self.den)
136    }
137
138    pub fn div_scalar<T: Into<Coef>>(&self, d: T) -> Self {
139        Self::new(self.num.clone(), self.den * d.into())
140    }
141
142    pub fn eval(&self, point: impl FnMut(Var) -> i128) -> (Coef, Coef) {
143        let n = self.num.eval(point);
144        let g = n.gcd(&self.den);
145        (n / g, self.den / g)
146    }
147
148    pub fn eval_int(&self, point: impl FnMut(Var) -> i128) -> Option<Coef> {
149        let (num, den) = self.eval(point);
150        if den == 1.into() { Some(num) } else { None }
151    }
152
153    pub fn substitute(&self, v: Var, q: &Poly) -> Self {
154        Self::new(self.num.substitute(v, q), self.den)
155    }
156
157    pub fn assert_canonical(&self) {
158        self.num.assert_canonical();
159        assert!(
160            self.den.is_positive(),
161            "denominator {:?} not positive",
162            self.den
163        );
164        assert_eq!(
165            self.num.coef_gcd().gcd(&self.den),
166            1.into(),
167            "fraction not reduced: content {:?} vs den {:?}",
168            self.num.coef_gcd(),
169            self.den
170        );
171    }
172
173    fn debug_checked(self) -> Self {
174        #[cfg(debug_assertions)]
175        self.assert_canonical();
176        self
177    }
178
179    pub fn sum_below(&self, var: Var, hi: &Poly) -> Self {
180        assert_eq!(hi.degree_in(var), 0);
181        let kmax = self.num().degree_in(var);
182        if kmax == 0 {
183            return Self::zero();
184        }
185
186        let kmax = kmax as usize;
187
188        let mut groups: Vec<Vec<(Coef, Mono)>> = vec![Vec::new(); kmax + 1];
189        for (c, m) in self.num().terms().iter().rev() {
190            let k = m.degree_in(var) as usize;
191            let rest = Mono::new(m.exps().iter().copied().filter(|&(v, _)| v != var));
192            groups[k].push((*c, rest));
193        }
194
195        let mut falling_factorials: Vec<Poly> = Vec::with_capacity(kmax + 1);
196        falling_factorials.push(hi.clone());
197        for j in 1..=kmax {
198            let factor = hi.sub(&Poly::constant(Coef::new(j as i128)));
199            falling_factorials.push(falling_factorials[j - 1].mul(&factor));
200        }
201
202        let s2 = stirling2(kmax);
203        let mut total = Self::zero();
204        for (k, group) in groups.into_iter().enumerate() {
205            let c_k = Poly::from_terms(group);
206            if c_k.is_zero() {
207                continue;
208            }
209
210            let mut a_k = Self::zero();
211            for (j, &s) in s2[k].iter().enumerate() {
212                if !s.is_zero() {
213                    a_k = a_k.add(&Self::new(
214                        falling_factorials[j].mul_scalar(s),
215                        Coef::new((j + 1) as i128),
216                    ));
217                }
218            }
219            total = total.add(&Self::new(c_k, 1).mul(&a_k));
220        }
221        total.div_scalar(self.den())
222    }
223
224    pub fn sum_range(&self, var: Var, lo: &Poly, hi: &Poly) -> Self {
225        self.sum_below(var, hi).sub(&self.sum_below(var, lo))
226    }
227}

229fn stirling2(kmax: usize) -> Vec<Vec<Coef>> {
230    let mut s2: Vec<Vec<Coef>> = Vec::with_capacity(kmax + 1);
231    s2.push(vec![Coef::new(1)]);
232    for k in 1..=kmax {
233        let mut row = Vec::with_capacity(k + 1);
234        row.push(Coef::new(0));
235        for j in 1..=k {
236            let above = if j < k { s2[k - 1][j] } else { Coef::new(0) };
237            let diag = s2[k - 1][j - 1];
238            row.push((Coef::new(j as i128) * above) + diag);
239        }
240        s2.push(row);
241    }
242    s2
243}

250    #[test]
251    fn ratio_reduce() {
252        let half_n = Count::new(poly!(2 * x), 4);
253        assert_eq!(half_n.num(), &poly!(x));
254        assert_eq!(half_n.den(), 2.into());
255        assert_eq!(Count::new(poly!(2 * x + 2), 2), Count::new(poly!(x + 1), 1));
256        assert_eq!(Count::new(poly!(2 * x), -1), Count::new(poly!(-2 * x), 1));
257    }
258
259    #[test]
260    fn arith_sanity() {
261        // N/2 * N/3 = N^2/6; N/2 − N/2 = 0
262        let (a, b) = (Count::new(poly!(x), 2), Count::new(poly!(x), 3));
263        assert_eq!(a.add(&b), Count::new(poly!(5 * x), 6));
264        assert_eq!(a.sub(&a), Count::new(poly!(), 1))
265    }

283    #[test]
284    fn faulhaber() {
285        // Σ_{0<=i<B} i^3 = B^2(B−1)^2/4
286        let cube = Count::new(poly!(i ^ 3), 1);
287        let s = cube.sum_below('i' as Var, &poly!(b));
288        let b2 = poly!(b ^ 2);
289        let bm1 = poly!(b - 1);
290        assert_eq!(s, Count::new(b2.mul(&bm1.pow(2)), 4));
291
292        // (Σ i)^2 = Σ i^3
293        let triangle = Count::new(poly!(i), 1).sum_below('i' as Var, &poly!(b));
294        assert_eq!(triangle.mul(&triangle), s);
295    }
