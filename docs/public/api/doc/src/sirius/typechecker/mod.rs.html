17pub fn check_source(source: &mut ParserOutput) -> Errors {
18    if let Some(tree) = &mut source.tree
19        && source.errors.is_empty()
20    {
21        let mut errors = vec![];
22
23        let mut global_types = Scope::default();
24        global_types.push();
25        let mut fail_check_signatures = false;
26
27        for fun in &tree.0 {
28            match fun_type(fun) {
29                Ok(t) => {
30                    global_types.insert(fun.name.data.clone(), Pos::new_at(t, &fun.name));
31                }
32                Err(e) => {
33                    errors.extend(e);
34                    fail_check_signatures = true;
35                }
36            }
37        }
38
39        if fail_check_signatures {
40            return errors;
41        }
42
43        for fun in tree.0.iter_mut() {
44            let p_args = Pos::inner_collect(&fun.type_args);
45            errors.extend(FunctionTypeChecker::run(&mut global_types, p_args, fun));
46        }
47
48        errors
49    } else {
50        vec![]
51    }
52}
53
54struct FunctionTypeChecker<'a> {
55    errors: Errors,
56    scope: &'a mut Scope<Pos<Type>>,
57    p_vars: Vec<String>,
58}
59
60impl<'a> FunctionTypeChecker<'a> {
61    pub fn run(
62        globals: &'a mut Scope<Pos<Type>>,
63        p_vars: Vec<String>,
64        ast: &'a mut Function,
65    ) -> Errors {
66        let mut checker = Self {
67            errors: vec![],
68            scope: globals,
69            p_vars,
70        };
71
72        checker.scope.push();
73        checker.traverse_block(&ast.body);
74        checker.scope.pop();
75
76        checker.errors
77    }
78
79    fn visit_expr(&mut self, _expr: &Expr) {}
80
81    fn visit_print(&mut self, expr: &Expr) {
82        self.visit_expr(expr);
83    }
84
85    fn visit_return(&mut self, expr: &Expr) {
86        self.visit_expr(expr);
87    }
88
89    fn visit_yield(&mut self, expr: &Expr) {
90        self.visit_expr(expr);
91    }
92
93    fn visit_yield_from(&mut self, expr: &Expr) {
94        self.visit_expr(expr);
95    }
96
97    fn visit_let(&mut self, _mutable: bool, _name: Pos<&str>, ann: Option<&Expr>, value: &Expr) {
98        if let Some(ann) = ann
99            && let Err(e) = annotation(ann, &self.p_vars)
100        {
101            self.errors.push(e);
102        }
103
104        self.visit_expr(value);
105    }
106
107    fn visit_assign(&mut self, _place: &Expr, _op: &AssnOp, value: &Expr) {
108        self.visit_expr(value);
109    }
110
111    fn visit_if(&mut self, cond: &Expr) {
112        self.visit_expr(cond);
113    }
114
115    fn visit_for(&mut self, _iter: Pos<&str>, lower: &Expr, upper: &Expr) {
116        self.visit_expr(lower);
117        self.visit_expr(upper);
118    }
119
120    fn traverse_block(&mut self, block: &Block) {
121        for stmt in block.stmts.iter() {
122            match &stmt.data {
123                S::Print(expr) => self.visit_print(expr),
124                S::Return(expr) => self.visit_return(expr),
125                S::Yield(expr) => self.visit_yield(expr),
126                S::YieldFrom(expr) => self.visit_yield_from(expr),
127                S::Let {
128                    mutable,
129                    name,
130                    ann,
131                    value,
132                } => self.visit_let(*mutable, name.as_ref(), ann.as_ref(), value),
133                S::Assign { place, op, value } => self.visit_assign(place, op, value),
134                S::If {
135                    cond,
136                    true_body,
137                    false_body,
138                } => {
139                    self.scope.push();
140                    self.visit_if(cond);
141                    self.scope.push();
142                    self.traverse_block(true_body);
143                    self.scope.pop();
144                    if let Some(false_body) = false_body {
145                        self.scope.push();
146                        self.traverse_block(false_body);
147                        self.scope.pop();
148                    }
149                    self.scope.pop();
150                }
151                S::For {
152                    iter,
153                    lower,
154                    upper,
155                    body,
156                } => {
157                    self.scope.push();
158                    self.visit_for(iter.as_ref(), lower, upper);
159                    self.traverse_block(body);
160                    self.scope.pop();
161                }
162            }
163        }
164    }
165}
