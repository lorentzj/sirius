8fn scalars(name: &str) -> Option<T> {
9    match name {
10        "f32" => Some(T::F32),
11        "i32" => Some(T::I32),
12        "bool" => Some(T::Bool),
13        "void" => Some(T::Void),
14        _ => None,
15    }
16}

18pub fn annotation(ann: &Expr, p_vars: &Vec<String>) -> Result<Type, Error> {
19    match &ann.data {
20        E::Ident(s) => {
21            if let Some(t) = scalars(s) {
22                Ok(Type::new_at(t, ann))
23            } else if let Some(p_position) = p_vars.iter().position(|v| v == s) {
24                Ok(Type::size(Poly::var(p_position as u32, 1), ann))
25            } else {
26                Err(ann.type_error(&format!("unknown type \"{}\"", s)))
27            }
28        }
29        E::Int(i) => Ok(Type::size(Poly::constant(*i as i128), ann)),
30        E::Tuple(inner) => {
31            let mut types = vec![];
32            for e in inner {
33                types.push(annotation(e, p_vars)?);
34            }
35            Ok(Type::new_at(T::Tuple(types), ann))
36        }
37        E::Access(t, dims) => {
38            let parsed_t = annotation(t, p_vars)?;
39            let mut parsed_dims = vec![];
40            for dim in dims {
41                match &dim.data {
42                    AD::Point(e) => {
43                        let p = annotation(e, p_vars)?;
44                        if let Type {
45                            data: T::Size(poly),
46                            ..
47                        } = p
48                        {
49                            parsed_dims.push(poly);
50                        } else {
51                            return Err(e.type_error("dimension must be a poly expression"));
52                        }
53                    }
54                    AD::Range(_, _) => {
55                        return Err(dim.type_error("ranges not supported in type annotations"));
56                    }
57                }
58            }
59            if let Type {
60                data:
61                    T::Array {
62                        elem: inner_t,
63                        shape: mut start_dims,
64                    },
65                ..
66            } = parsed_t
67            {
68                // array dimensions are cumulative
69                // so extend the existing dimensions with the new ones
70                // e.g. a[10][20] == a[10, 20]
71                start_dims.extend(parsed_dims);
72                Ok(Type::new_at(
73                    T::Array {
74                        elem: Box::new(*inner_t),
75                        shape: start_dims,
76                    },
77                    ann,
78                ))
79            } else {
80                Ok(Type::new_at(
81                    T::Array {
82                        elem: Box::new(parsed_t),
83                        shape: parsed_dims,
84                    },
85                    ann,
86                ))
87            }
88        }
89        E::BinOp(lhs, op, rhs) => {
90            let lhs_t = annotation(lhs, p_vars)?;
91            let rhs_t = annotation(rhs, p_vars)?;
92
93            let (lhs_poly, rhs_poly) = match (lhs_t.data, rhs_t.data) {
94                (T::Size(l), T::Size(r)) => (l, r),
95                (T::Size(_), _) => {
96                    return Err(rhs.type_error("operand must be a poly expression"));
97                }
98                (_, _) => {
99                    return Err(lhs.type_error("operand must be a poly expression"));
100                }
101            };
102
103            match op {
104                Op::Add => Ok(Type::size(lhs_poly.add(&rhs_poly), ann)),
105                Op::Sub => Ok(Type::size(lhs_poly.sub(&rhs_poly), ann)),
106                Op::Mul => Ok(Type::size(lhs_poly.mul(&rhs_poly), ann)),
107                Op::Exp => match rhs_poly.as_constant() {
108                    Some(rhs_const) => {
109                        if rhs_const.is_zero() {
110                            Ok(Type::size(Poly::constant(1), ann))
111                        } else if !rhs_const.is_positive() {
112                            Err(rhs.type_error("power must be non-negative"))
113                        } else {
114                            Ok(Type::size(lhs_poly.pow(rhs_const.get() as u32), ann))
115                        }
116                    }
117                    None => Err(rhs.type_error("power must be a constant poly expression")),
118                },
119                _ => Err(ann.type_error(&format!(
120                    "invalid binary operation \"{op:?}\" in type annotation"
121                ))),
122            }
123        }
124        E::FnCall(fun, p_args, args) => match &fun.data {
125            E::Ident(ident) => {
126                if ident == "Ind" {
127                    if !p_args.is_empty() {
128                        return Err(ann.type_error("cannot pass typevar here"));
129                    }
130
131                    if args.len() != 1 {
132                        return Err(ann.type_error("Ind expects one argument"));
133                    }
134
135                    let inner_t = annotation(&args[0], p_vars)?;
136
137                    if let T::Size(p) = inner_t.data {
138                        Ok(Type::ind(p.clone(), ann))
139                    } else {
140                        Err(ann.type_error("Ind expects size argument"))
141                    }
142                } else {
143                    Err(ann.type_error("invalid type annotation"))
144                }
145            }
146            _ => Err(ann.type_error("invalid type annotation")),
147        },
148        _ => Err(ann.type_error("invalid type annotation")),
149    }
150}

152pub fn fun_type(fun: &Function) -> Result<Type, Errors> {
153    let p_vars = Pos::inner_collect(&fun.type_args);
154
155    let mut args = vec![];
156    let mut p_constraints = vec![];
157    let mut errors = vec![];
158
159    for (name, ann) in fun.type_constraints.iter() {
160        match annotation(ann, &p_vars) {
161            Ok(Type {
162                data: T::Size(p),
163                end,
164                ..
165            }) => {
166                p_constraints.push(Pos::new(name.start, (name.data.clone(), p), end));
167            }
168            Ok(_) => {
169                errors.push(ann.type_error("constraint must be a poly expression"));
170            }
171            Err(e) => {
172                errors.push(e);
173            }
174        }
175    }
176
177    for (_, ann) in fun.args.iter() {
178        match annotation(ann, &p_vars) {
179            Ok(t) => {
180                args.push(t);
181            }
182            Err(e) => {
183                errors.push(e);
184            }
185        }
186    }
187
188    for (_, ann) in fun.args.iter() {
189        match annotation(ann, &p_vars) {
190            Ok(t) => {
191                args.push(t);
192            }
193            Err(e) => {
194                errors.push(e);
195            }
196        }
197    }
198
199    let ret_t = match &fun.ret {
200        Some(ret) => match annotation(ret, &p_vars) {
201            Ok(t) => t,
202            Err(e) => {
203                errors.push(e);
204                Type::new_at(T::Void, ret)
205            }
206        },
207        None => Type::new_at(T::Void, &fun.name),
208    };
209
210    Ok(Type::new_fn(fun.type_args.clone(), args, ret_t, &fun.name))
211}
