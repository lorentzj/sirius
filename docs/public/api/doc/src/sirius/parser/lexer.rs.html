196fn parse_keyword(s: &str) -> Option<Tok> {
197    match s {
198        "let" => Some(Tok::Keyword(Keyword::Let)),
199        "print" => Some(Tok::Keyword(Keyword::Print)),
200        "if" => Some(Tok::Keyword(Keyword::If)),
201        "else" => Some(Tok::Keyword(Keyword::Else)),
202        "true" => Some(Tok::Keyword(Keyword::True)),
203        "false" => Some(Tok::Keyword(Keyword::False)),
204        "and" => Some(Tok::Keyword(Keyword::And)),
205        "or" => Some(Tok::Keyword(Keyword::Or)),
206        "fn" => Some(Tok::Keyword(Keyword::Fn)),
207        "return" => Some(Tok::Keyword(Keyword::Return)),
208        "for" => Some(Tok::Keyword(Keyword::For)),
209        "from" => Some(Tok::Keyword(Keyword::From)),
210        "to" => Some(Tok::Keyword(Keyword::To)),
211        "st" => Some(Tok::Keyword(Keyword::SuchThat)),
212        "\\in" => Some(Tok::Keyword(Keyword::In)),

214        "mut" => Some(Tok::Keyword(Keyword::Mut)),

219fn lex_substring_alpha_num_dec(s: &str, line: usize, start: usize, end: usize) -> Vec<Token> {
220    match parse_keyword(s) {
221        Some(tok) => vec![Token::new(tok, line, start, end)],
222        None => {
223            if s == "." {
224                vec![Token::new(Tok::Op(Op::Dot), line, start, end)]
225            } else {
226                match s.chars().next() {
227                    Some('0'..='9') => {
228                        if s.contains('.') {
229                            let parsed_number = String::from_iter(s.chars()).parse::<f64>();
230                            match parsed_number {
231                                Ok(n) => {
232                                    if n.is_infinite() {
233                                        vec![Token::new(
234                                            Tok::Error("number overflowed f64".into()),
235                                            line,
236                                            start,
237                                            end,
238                                        )]
239                                    } else {
240                                        vec![Token::new(Tok::Float(n), line, start, end)]
241                                    }
242                                }
243                                Err(msg) => {
244                                    vec![Token::new(Tok::Error(msg.to_string()), line, start, end)]
245                                }
246                            }
247                        } else {
248                            let parsed_number = String::from_iter(s.chars()).parse::<i64>();
249                            match parsed_number {
250                                Ok(n) => vec![Token::new(Tok::Int(n), line, start, end)],
251                                Err(msg) => {
252                                    vec![Token::new(Tok::Error(msg.to_string()), line, start, end)]
253                                }
254                            }
255                        }
256                    }
257                    None => vec![],
258                    _ => {
259                        if s.contains('.') {
260                            let mut new_start = start;
261                            let mut res = vec![];
262                            for sub in s.split('.') {
263                                res.extend(lex_substring_alpha_num_dec(
264                                    sub,
265                                    line,
266                                    new_start,
267                                    new_start + sub.len(),
268                                ));
269                                new_start += sub.len() + 1;
270                                res.push(Token::new(
271                                    Tok::Op(Op::Dot),
272                                    line,
273                                    new_start - 1,
274                                    new_start,
275                                ));
276                            }
277                            res.pop();
278                            res
279                        } else {
280                            vec![Token::new(Tok::Identifier(s.to_string()), line, start, end)]
281                        }
282                    }
283                }
284            }
285        }
286    }
287}

289pub fn tokenize(code: &str) -> Vec<Token> {
290    let mut tokens: Vec<Token> = vec![];
291    let mut bracket_level = (0, 0, 0);
292    let mut indent_spaces: Option<usize> = Some(0);
293    let mut commenting = false;
294    let mut prev_indent_level = 0;
295    let mut line = 0;
296    let mut col = 0;
297
298    let mut curr_substring_start: Option<usize> = None;
299
300    for (i, char) in code.char_indices() {
301        if commenting {
302            if char == '\n' {
303                commenting = false;
304            } else {
305                if let Some(Token {
306                    data: Tok::Comment,
307                    end,
308                    ..
309                }) = tokens.last_mut()
310                {
311                    *end += 1;
312                }
313                col += 1;
314                continue;
315            }
316        }
317
318        match char {
319            '0'..='9' | 'A'..='Z' | 'a'..='z' | '\u{0370}'..='\u{03FF}' | '_' | '.' | '\\' => {
320                if let Some(spaces) = indent_spaces {
321                    if spaces % 4 == 0 {
322                        let curr_indent_level = spaces / 4;
323                        #[allow(clippy::comparison_chain)]
324                        if curr_indent_level > prev_indent_level {
325                            for c in prev_indent_level..curr_indent_level {
326                                tokens.push(Token::new(Tok::Indent, line, c * 4, (c + 1) * 4));
327                            }
328                        } else if prev_indent_level > curr_indent_level {
329                            for _c in curr_indent_level..prev_indent_level {
330                                tokens.push(Token::new(
331                                    Tok::Dedent,
332                                    line,
333                                    0,
334                                    curr_indent_level * 4,
335                                ));
336                            }
337                        }
338                        prev_indent_level = curr_indent_level;
339                    } else {
340                        tokens.push(Token::new(
341                            Tok::IndentError(format!(
342                                "indent level must be multiple of 4; found {spaces}"
343                            )),
344                            line,
345                            col - spaces,
346                            col,
347                        ));
348                    }
349                }
350                indent_spaces = None;
351                if curr_substring_start.is_none() {
352                    curr_substring_start = Some(i);
353                }
354
355                col += 1;
356            }
357            _ => {
358                if let Some(curr_ss) = curr_substring_start {
359                    tokens.extend(lex_substring_alpha_num_dec(
360                        &code[curr_ss..i],
361                        line,
362                        col - (i - curr_ss),
363                        col,
364                    ));
365
366                    curr_substring_start = None;
367                }
368
369                if char == '\n' {
370                    line += 1;
371                    col = 0;
372                } else {
373                    if char != ' ' && char != '\u{00A0}' {
374                        if let Some(spaces) = indent_spaces {
375                            if spaces % 4 == 0 {
376                                let curr_indent_level = spaces / 4;
377                                #[allow(clippy::comparison_chain)]
378                                if curr_indent_level > prev_indent_level {
379                                    for c in prev_indent_level..curr_indent_level {
380                                        tokens.push(Token::new(
381                                            Tok::Indent,
382                                            line,
383                                            c * 4,
384                                            (c + 1) * 4,
385                                        ));
386                                    }
387                                } else if prev_indent_level > curr_indent_level {
388                                    for _c in curr_indent_level..prev_indent_level {
389                                        tokens.push(Token::new(
390                                            Tok::Dedent,
391                                            line,
392                                            0,
393                                            curr_indent_level * 4,
394                                        ));
395                                    }
396                                }
397                                prev_indent_level = curr_indent_level;
398                            } else {
399                                tokens.push(Token::new(
400                                    Tok::IndentError(format!(
401                                        "indent level must be multiple of 4; found {spaces}"
402                                    )),
403                                    line,
404                                    col - spaces,
405                                    col,
406                                ));
407                            }
408                        }
409                        indent_spaces = None;
410                    }
411                    col += 1;
412                }
413
414                let mut should_pop = false;
415                let mut clear_line_whitespace = false;
416
417                let new_token = match char {
418                    ' ' | '\u{00A0}' => match indent_spaces {
419                        Some(i) => {
420                            indent_spaces = Some(i + 1);
421                            continue;
422                        }
423                        None => continue,
424                    },
425                    '\n' => {
426                        if bracket_level == (0, 0, 0) {
427                            indent_spaces = Some(0);
428                            Token::new(Tok::NewLine, line, col, col)
429                        } else {
430                            continue;
431                        }
432                    }
433                    ';' => Token::new(Tok::Semicolon, line, col - 1, col),
434                    ':' => Token::new(Tok::Colon, line, col - 1, col),
435                    '(' => {
436                        bracket_level.0 += 1;
437                        Token::new(Tok::OpenParen, line, col - 1, col)
438                    }
439                    ')' => {
440                        bracket_level.0 -= 1;
441                        Token::new(Tok::CloseParen, line, col - 1, col)
442                    }
443                    '{' => {
444                        bracket_level.1 += 1;
445                        Token::new(Tok::OpenCurly, line, col - 1, col)
446                    }
447                    '}' => {
448                        bracket_level.1 -= 1;
449                        Token::new(Tok::CloseCurly, line, col - 1, col)
450                    }
451                    '[' => {
452                        bracket_level.2 += 1;
453                        Token::new(Tok::OpenSqBracket, line, col - 1, col)
454                    }
455                    ']' => {
456                        bracket_level.2 -= 1;
457                        Token::new(Tok::CloseSqBracket, line, col - 1, col)
458                    }
459                    '#' => Token::new(Tok::Pound, line, col - 1, col),
460                    '.' => Token::new(Tok::Op(Op::Dot), line, col - 1, col),
461                    '^' => Token::new(Tok::Op(Op::Exp), line, col - 1, col),
462                    '*' => Token::new(Tok::Op(Op::Mul), line, col - 1, col),
463                    '/' => {
464                        if let Some(Token {
465                            data: Tok::Op(Op::Div),
466                            start,
467                            ..
468                        }) = tokens.last()
469                        {
470                            should_pop = true;
471                            clear_line_whitespace = true;
472                            commenting = true;
473                            Token::new(Tok::Comment, line, *start, col)
474                        } else {
475                            Token::new(Tok::Op(Op::Div), line, col - 1, col)
476                        }
477                    }
478                    '+' => Token::new(Tok::Op(Op::Add), line, col - 1, col),
479                    '-' => Token::new(Tok::Op(Op::Sub), line, col - 1, col),
480                    '!' => Token::new(Tok::Op(Op::Not), line, col - 1, col),
481                    '\'' => Token::new(Tok::Op(Op::Tick), line, col - 1, col),
482                    '>' => {
483                        if let Some(Token {
484                            data: Tok::Op(Op::Sub),
485                            start,
486                            ..
487                        }) = tokens.last()
488                        {
489                            should_pop = true;
490                            Token::new(Tok::Op(Op::Apply), line, *start, col)
491                        } else {
492                            Token::new(Tok::Op(Op::Greater), line, col - 1, col)
493                        }
494                    }
495                    '<' => Token::new(Tok::Op(Op::Less), line, col - 1, col),
496                    '?' => Token::new(Tok::Op(Op::Option), line, col - 1, col),
497                    '=' => {
498                        if let Some(Token {
499                            data: Tok::AssignOp(AssnOp::Eq),
500                            start,
501                            ..
502                        }) = tokens.last()
503                        {
504                            should_pop = true;
505                            Token::new(Tok::Op(Op::Eq), line, *start, col)
506                        } else if let Some(Token {
507                            data: Tok::Op(last_op),
508                            start,
509                            ..
510                        }) = tokens.last()
511                        {
512                            match last_op {
513                                Op::Greater => {
514                                    should_pop = true;
515                                    Token::new(Tok::Op(Op::GreaterOrEq), line, *start, col)
516                                }
517                                Op::Less => {
518                                    should_pop = true;
519                                    Token::new(Tok::Op(Op::LessOrEq), line, *start, col)
520                                }
521                                Op::Not => {
522                                    should_pop = true;
523                                    Token::new(Tok::Op(Op::NotEq), line, *start, col)
524                                }
525                                Op::Add => {
526                                    should_pop = true;
527                                    Token::new(Tok::AssignOp(AssnOp::Add), line, *start, col)
528                                }
529                                Op::Sub => {
530                                    should_pop = true;
531                                    Token::new(Tok::AssignOp(AssnOp::Sub), line, *start, col)
532                                }
533                                Op::Mul => {
534                                    should_pop = true;
535                                    Token::new(Tok::AssignOp(AssnOp::Mul), line, *start, col)
536                                }
537                                Op::Div => {
538                                    should_pop = true;
539                                    Token::new(Tok::AssignOp(AssnOp::Div), line, *start, col)
540                                }
541                                _ => Token::new(Tok::AssignOp(AssnOp::Eq), line, col - 1, col),
542                            }
543                        } else {
544                            Token::new(Tok::AssignOp(AssnOp::Eq), line, col - 1, col)
545                        }
546                    }
547                    ',' => Token::new(Tok::Op(Op::Comma), line, col - 1, col),
548                    _ => Token::new(
549                        Tok::Error(format!("unknown token \"{char}\"")),
550                        line,
551                        col - 1,
552                        col,
553                    ),
554                };
555
556                if should_pop {
557                    tokens.pop();
558                }
559
560                if clear_line_whitespace {
561                    while let Some(last) = tokens.last() {
562                        if last.line == line {
563                            match last.data {
564                                Tok::Indent => prev_indent_level -= 1,
565                                Tok::Dedent => prev_indent_level += 1,
566                                Tok::IndentError(_) => (),
567                                _ => break,
568                            }
569                            tokens.pop();
570                        } else {
571                            break;
572                        }
573                    }
574                }
575
576                tokens.push(new_token);
577            }
578        }
579    }
580
581    if let Some(curr_ss) = curr_substring_start {
582        tokens.extend(lex_substring_alpha_num_dec(
583            &code[curr_ss..code.len()],
584            line,
585            col - (code.len() - curr_ss),
586            col,
587        ));
588    }
589
590    if prev_indent_level > 0 {
591        tokens.push(Token::new(Tok::NewLine, line, col, col));
592
593        for _i in 0..prev_indent_level {
594            tokens.push(Token::new(Tok::Dedent, line, col, col))
595        }
596    }
597
598    tokens
599}
